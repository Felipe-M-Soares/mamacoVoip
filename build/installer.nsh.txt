!macro customInit
  ; Verifica se uma versão anterior está instalada e pergunta se o usuário quer atualizar
  ReadRegStr $R0 HKLM "Software\Microsoft\Windows\CurrentVersion\Uninstall\${APP_GUID}" "DisplayVersion"
  ${If} $R0 != ""
    MessageBox MB_YESNO|MB_ICONQUESTION "Uma versão anterior (${PRODUCT_VERSION}) está instalada. Deseja atualizar?" IDYES true IDNO false
    true:
      Goto +2
    false:
      Quit
  ${EndIf}
!macroend

!macro customInstall
  ; Remove atalhos antigos se existirem (para evitar duplicação)
  Delete "$DESKTOP\${PRODUCT_NAME}.lnk"
  Delete "$SMPROGRAMS\${PRODUCT_NAME}\*.lnk"
!macroend
